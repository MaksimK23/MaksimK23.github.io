# Шаблон Telegram Instant View для сайта «Красный Кабан»
# Скопируйте целиком в поле RULES на instantview.telegram.org

~version: "2.1"

# Только страницы статей
?exists: //article//div[has-class("prose")]

# Заголовок и тело статьи
title:     //article//header/h1
body:      //article//div[has-class("prose")]
subtitle:  //article//header/p[has-class("article-lede")]

# Автор и дата
author:         "Красный Кабан"
published_date: //article//header//time/@datetime

# ── Формулы LaTeX ──
# На сайте формулы нарисованы KaTeX, а для Telegram рядом лежат PNG-картинки.
# Убираем разметку KaTeX и включаем картинки.
@remove: //*[has-class("math-web")]
@set_attr(src, @data-src): //img[has-class("math-iv")]
# Формулы внутри текста — маленькие картинки в строке
@replace_tag(<pic>): //span[has-class("math-inline")]/img
# Отдельные формулы выносим из абзаца
@split_parent: //p/figure[has-class("math-block")]
# Обычные отдельные формулы — в натуральном размере, своим абзацем
@replace_tag(<pic>): //figure[has-class("math-pic")]/img
@replace_tag(<p>): //figure[has-class("math-pic")]
# Очень широкие формулы остаются картинкой на всю ширину экрана

# ── Служебные элементы сайта ──
@remove: //a[has-class("h-anchor")]
@remove: //button[has-class("yt-facade")]
@remove: //div[has-class("mini-bar")]
# Видео: на сайте до клика показывается обложка, для Telegram возвращаем адрес плеера
@set_attr(src, @data-src): //iframe[@data-src]
# Заголовки «Блок N»: номер и время отдельной строкой
@after(" · "): //span[has-class("block-num")]
@after(". "): //span[has-class("block-kicker")]
# Выделенные блоки (тезис, факт, критика) — как цитаты
@replace_tag(<blockquote>): //aside[has-class("callout")]

# ── Видео YouTube ──
@wrap(<figure>): //div[has-class("video-wrap")]/iframe
@replace_tag(<video>): //figure/iframe[contains(@src, "youtube.com") or contains(@src, "youtu.be")]
